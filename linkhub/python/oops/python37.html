# Ek Python program likho jo class method ka use karke
# car ki speed print kare.

# class create karna
class Car:

    # method create karna
    def speed(self, s):

        # speed print karna
        print("Speed :", s)

# user se speed input lena
speed = int(input("Enter Your Car Speed : "))

# object create karna
c = Car()

# method call karna
c.speed(speed)
📊 Output:

Enter Your Car Speed : 120
Speed : 120
