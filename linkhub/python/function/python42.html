# Ek Python program likho jo function ka use karke
# kisi string me vowels ki total counting kare.

# function create karna
def vowel(s):

    # vowel counter variable
    count = 0

    # string ke characters par loop chalana
    for i in s:

        # vowel check karna
        if i in "aeiouAEIOU":

            count += 1

    # total vowels print karna
    print("Vowels in Given String :", count)

# user se text input lena
s = input("Enter Any Text : ")

# function call karna
vowel(s)
📊 Output:

Enter Any Text : Shivam Kumar
Vowels in Given String : 4
