# Question 33:
# Ek Python program likho jo constructor ka use karke
# book ka name aur author name print kare.

# class create karna
class Book:

    # constructor create karna
    def __init__(self, book_name, author):

        # instance variables create karna
        self.book_name = book_name
        self.author = author

# object create karna
b = Book("Python", "Shivam")

# book name print karna
print("Book Name :", b.book_name)

# author name print karna
print("Author Name :", b.author)
📊 Output:

Book Name : Python
Author Name : Shivam
