# Ek Python program likho jo recursion function ka use karke
# kisi number ka factorial calculate kare.

# recursive function create karna
def fact(n):

    # base condition check karna
    if n == 1 or n == 0:

        return 1

    else:

        # recursive factorial calculate karna
        return n * fact(n - 1)

# user se number input lena
n = int(input("Enter a Number : "))

# factorial calculate karna
factorial = fact(n)

# final output print karna
print("Factorial of Given Number :", factorial)
📊 Output:

Enter a Number : 5
Factorial of Given Number : 120
