# Ek Python program likho jo constructor ka use karke
# car ka brand name print kare.

# class create karna
class Car:

    # constructor create karna
    def __init__(self, brand):

        # instance variable create karna
        self.brand = brand

# object create karna
c = Car("BMW")

# car brand print karna
print("Car Brand :", c.brand)
📊 Output:

Car Brand : BMW
