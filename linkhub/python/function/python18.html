# Ek Python program likho jo function ka use karke
# Celsius temperature ko Fahrenheit me convert kare.

# function create karna
def temp(c):

    # Celsius ko Fahrenheit me convert karna
    print("Temperature in Fahrenheit :", (c * 9 / 5) + 32)

# heading print karna
print("\n===== CELSIUS TO FAHRENHEIT =====\n")

# user se temperature input lena
c = float(input("Enter Temperature in Celsius : "))

# function call karna
temp(c)
📊 Output:

===== CELSIUS TO FAHRENHEIT =====

Enter Temperature in Celsius : 37
Temperature in Fahrenheit : 98.6
