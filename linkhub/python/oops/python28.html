# ==========================================
# Student Result Management System
# ==========================================


# class create karna
class Student:

    # constructor create karna
    def __init__(self, name, marks, age, college):

        # instance variables create karna
        self.name = name
        self.marks = marks
        self.age = age
        self.college = college


    # result method create karna
    def result(self):

        # pass or fail check karna
        if self.marks >= 33:

            print("\nResult : Pass")

        else:

            print("\nResult : Fail")


# ==========================================
# Object Creation
# ==========================================

# object create karna
Student1 = Student("Shivam", 89, 21, "NIET")


# ==========================================
# Student Information
# ==========================================

print("\n========== STUDENT RECORD ==========\n")

print("Name    :", Student1.name)
print("Marks   :", Student1.marks)
print("Age     :", Student1.age)
print("College :", Student1.college)


# ==========================================
# Result
# ==========================================

# result method call karna
Student1.result()
