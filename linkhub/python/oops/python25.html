# Question 25:
# Ek Python program likho jo operator overloading ka use karke
# do objects ko add kare.

# class create karna
class Number:

    # constructor create karna
    def __init__(self, x):

        # instance variable create karna
        self.x = x

    # operator overloading method create karna
    def __add__(self, other):

        # objects ki values add karna
        return self.x + other.x

# first object create karna
n1 = Number(10)

# second object create karna
n2 = Number(20)

# objects ko add karke result print karna
print(n1 + n2)
📊 Output:

30
