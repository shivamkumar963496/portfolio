# Ek Python program likho jo recursion function ka use karke
# 1 se n tak numbers ka sum calculate kare.

# recursive function create karna
def total(n):

    # base condition check karna
    if n == 1:

        return 1

    else:

        # recursive sum calculate karna
        return n + total(n - 1)

# user se number input lena
n = int(input("Enter a Number : "))

# total sum calculate karna
sum = total(n)

# final output print karna
print("Sum of Given Number :", sum)
📊 Output:

Enter a Number : 5
Sum of Given Number : 15
