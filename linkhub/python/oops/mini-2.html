# Ek Python mini project likho jo bank account management system banaye.
# Program me deposit, withdraw aur balance check system hona chahiye.


# =====================================
# Bank Account Management System
# =====================================


# class create karna
class BankAccount:

    # =================================
    # Constructor
    # =================================

    def __init__(self, name, balance):

        # instance variables create karna
        self.name = name
        self.__balance = balance


    # =================================
    # Deposit Method
    # =================================

    def deposit(self, amount):

        # deposit amount check karna
        if amount > 0:

            # balance update karna
            self.__balance += amount

            print(amount, "Deposited Successfully")

        else:

            print("Invalid Deposit Amount")


    # =================================
    # Withdraw Method
    # =================================

    def withdraw(self, amount):

        # sufficient balance check karna
        if amount <= self.__balance:

            # balance update karna
            self.__balance -= amount

            print(amount, "Withdraw Successfully")

        else:

            print("Insufficient Balance")


    # =================================
    # Get Balance Method
    # =================================

    def get_balance(self):

        # current balance return karna
        return self.__balance


# =====================================
# Create Account
# =====================================

# object create karna
acc1 = BankAccount("Shivam", 10000)


# =====================================
# Current Balance
# =====================================

print("Current Balance :", acc1.get_balance())


# =====================================
# Deposit Money
# =====================================

# deposit method call karna
acc1.deposit(5000)

print("After Deposit :", acc1.get_balance())


# =====================================
# Withdraw Money
# =====================================

# withdraw method call karna
acc1.withdraw(2000)

print("After Withdraw :", acc1.get_balance())
📊 Output:

Current Balance : 10000

5000 Deposited Successfully
After Deposit : 15000

2000 Withdraw Successfully
After Withdraw : 13000
