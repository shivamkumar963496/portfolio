# Question:
# Python me ek Library Management System program likhiye jo
# user ko book add, view, issue, return aur search karne ki
# facility de.
# -----------------------------------------------------------

# library books store karne ke liye list
library = []

# menu loop
while True:

    print("\n===== LIBRARY MANAGEMENT SYSTEM =====")

    print("1. Add Book")
    print("2. View Books")
    print("3. Issue Book")
    print("4. Return Book")
    print("5. Search Book")
    print("6. Exit")

    choice = input("Enter Your Choice : ")

    # Add Book
    if choice == "1":

        book = input("Enter Book Name : ")

        library.append(book)

        print("Book Added Successfully")

    # View Books
    elif choice == "2":

        if len(library) == 0:

            print("No Books Available")

        else:

            print("\nAvailable Books:")

            for i, book in enumerate(library, start=1):

                print(i, ".", book)

    # Issue Book
    elif choice == "3":

        book = input("Enter Book Name To Issue : ")

        found = False

        for item in library:

            if item.lower() == book.lower():

                library.remove(item)

                print("Book Issued Successfully")

                found = True
                break

        if not found:

            print("Book Not Found")

    # Return Book
    elif choice == "4":

        book = input("Enter Return Book Name : ")

        library.append(book)

        print("Book Returned Successfully")

    # Search Book
    elif choice == "5":

        book = input("Enter Book Name To Search : ")

        found = False

        for item in library:

            if item.lower() == book.lower():

                found = True
                break

        if found:

            print("Book Available")

        else:

            print("Book Not Available")

    # Exit Program
    elif choice == "6":

        print("Thank You Bye Bye")

        break

    # Invalid Choice
    else:

        print("Invalid Input")

2. Add Book option new book add karta hai.
3. View Books option saari available books dikhata hai.
4. Issue Book option book ko library se remove karta hai.
5. Return Book option issued book ko wapas add karta hai.
6. Search Book option book available hai ya nahi check karta hai.
7. Program menu driven hai aur loop ke through continuously chalta hai.
