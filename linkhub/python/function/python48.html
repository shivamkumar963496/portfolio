# Question 48:
# Ek Python program likho jo function ka use karke
# do numbers par basic calculator operations perform kare.

# function create karna
def cal(a, b):

    # addition print karna
    print("Sum of Given Numbers :", a + b)

    # subtraction print karna
    print("Subtraction of Given Numbers :", a - b)

    # multiplication print karna
    print("Multiplication of Given Numbers :", a * b)

    # division print karna
    print("Division of Given Numbers :", a / b)

# user se numbers input lena
m = int(input("Enter 1st Number : "))
n = int(input("Enter 2nd Number : "))

# function call karna
cal(m, n)
📊 Output:

Enter 1st Number : 20
Enter 2nd Number : 10

Sum of Given Numbers : 30
Subtraction of Given Numbers : 10
Multiplication of Given Numbers : 200
Division of Given Numbers : 2.0
