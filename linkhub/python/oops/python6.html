# Ek Python program likho jo class variable ka use karke
# sabhi students ka same college name print kare.

# class create karna
class College:

    # class variable create karna
    college_name = "NEELKANTH INSTITUTE OF ENGINEERING & TECHNOLOGY"

# first object create karna
s1 = College()

# second object create karna
s2 = College()

# first student ka college name print karna
print("Student-1 College Name :", s1.college_name)

# second student ka college name print karna
print("Student-2 College Name :", s2.college_name)
📊 Output:

Student-1 College Name : NEELKANTH INSTITUTE OF ENGINEERING & TECHNOLOGY
Student-2 College Name : NEELKANTH INSTITUTE OF ENGINEERING & TECHNOLOGY
