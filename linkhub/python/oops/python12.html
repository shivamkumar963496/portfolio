# Question 12:
# Ek Python program likho jo inheritance ka use karke
# parent class method ko child class se access kare.

# parent class create karna
class Parent:

    # parent class method create karna
    def show():

        # message print karna
        print("Parent Class")

# child class create karna
class Child(Parent):

    pass

# child class ko variable me store karna
c = Child

# inherited method call karna
c.show()
📊 Output:

Parent Class
