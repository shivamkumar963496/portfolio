# Question 38:
# Ek Python program likho jo constructor ka use karke
# laptop ka brand aur RAM print kare.

# class create karna
class Laptop:

    # constructor create karna
    def __init__(self, brand, ram):

        # instance variables create karna
        self.brand = brand
        self.ram = ram

# object create karna
l = Laptop("Dell", "16GB")

# laptop brand print karna
print(l.brand)

# laptop RAM print karna
print(l.ram)
📊 Output:

Dell
16GB
