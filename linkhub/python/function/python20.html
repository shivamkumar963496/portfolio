# Question 20:
# Ek Python program likho jo function ka use karke
# do numbers ko swap kare.

# function create karna
def swap(a, b):

    # swapping karna
    a, b = b, a

    # swapped values print karna
    print("After Swapping :")
    print("a =", a)
    print("b =", b)

# user se input lena
a = int(input("Enter Value of a : "))
b = int(input("Enter Value of b : "))

# function call karna
swap(a, b)
📊 Output:

Enter Value of a : 10
Enter Value of b : 20

After Swapping :
a = 20
b = 10
