# Ek Python program likho jo function ka use karke
# check kare ki candidate voting ke liye eligible hai ya nahi.

# function create karna
def check_age(age):

    # age check karna
    if age >= 18:

        print("ELIGIBLE FOR VOTING")

    else:

        print("NOT ELIGIBLE FOR VOTING")

# heading print karna
print("===== CHECK VOTING ELIGIBILITY =====")

# user se age input lena
age = int(input("Enter Candidate Age : "))

# function call karna
check_age(age)
