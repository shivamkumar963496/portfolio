🏦 Python Mini Project 7 - Bank Management System

# -----------------------------------------------------------
# Question:
# Python me ek Simple Bank Management System program likhiye
# jo user ko Deposit, Withdraw, Balance Check aur Exit ki
# facility de.
# -----------------------------------------------------------

# account ka initial balance
balance = 1000.0

# welcome message display karna
print("\n===== BANK MANAGEMENT SYSTEM =====")

# loop tab tak chalega jab tak user exit na kare
while True:

    print("\n1. Deposit")
    print("2. Withdraw")
    print("3. Check Balance")
    print("4. Exit")

    # user se choice lena
    choice = input("Enter Your Choice (1-4) : ")

    # Deposit Amount
    if choice == "1":

        amount = float(input("Enter Deposit Amount : ₹"))

        if amount > 0:

            balance += amount

            print("Deposit Successful")
            print("Current Balance : ₹", balance)

        else:

            print("Deposit Amount Must Be Greater Than 0")

    # Withdraw Amount
    elif choice == "2":

        amount = float(input("Enter Withdrawal Amount : ₹"))

        if amount <= 0:

            print("Withdrawal Amount Must Be Greater Than 0")

        elif amount <= balance:

            balance -= amount

            print("Withdrawal Successful")
            print("Current Balance : ₹", balance)

        else:

            print("Insufficient Balance")

    # Check Balance
    elif choice == "3":

        print("Available Balance : ₹", balance)

    # Exit Program
    elif choice == "4":

        print("\nTHANK YOU FOR BANKING WITH US")
        print("Visit Again")

        break

    # Invalid Choice
    else:

        print("Invalid Choice")
        print("Please Enter A Number Between 1 To 4")

6. Menu driven system loop ke through baar-baar chalta hai.
