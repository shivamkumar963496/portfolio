# Ek Python program likho jo function ka use karke
# Simple Interest calculate kare.

# function create karna
def si(p, r, t):

    # simple interest calculate karna
    print((p * r * t) / 100)

# user se input lena
p = int(input("Enter Principal Amount : "))
r = int(input("Enter Rate of Interest : "))
t = int(input("Enter Time in Months : "))

# function call karna
si(p, r, t)
