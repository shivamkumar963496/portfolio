# Question 26:
# Ek Python program likho jo recursion function ka use karke
# countdown print kare.

# recursive function create karna
def count(n):

    # base condition check karna
    if n == 0:

        return

    else:

        # current number print karna
        print(n)

        # function ko dubara call karna
        count(n - 1)

# function call karna
count(5)

5
4
3
2
1
