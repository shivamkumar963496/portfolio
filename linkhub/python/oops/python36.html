# Ek Python program likho jo class aur method ka use karke
# student ka grade calculate kare.

# class create karna
class Student:

    # constructor create karna
    def __init__(self, marks):

        # instance variable create karna
        self.marks = marks

    # grade method create karna
    def grade(self):

        # grade check karna
        if self.marks >= 90:

            print("Grade A")

        elif self.marks >= 75:

            print("Grade B")

        elif self.marks >= 40:

            print("Grade C")

        else:

            print("Fail")

# user se marks input lena
m = int(input("Enter Your Marks : "))

# object create karna
s = Student(m)

# grade method call karna
s.grade()
📊 Output:

Enter Your Marks : 85
Grade B
