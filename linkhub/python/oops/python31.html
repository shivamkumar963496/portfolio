# Ek Python program likho jo constructor aur method ka use karke
# ATM balance check kare.

# class create karna
class Atm:

    # constructor create karna
    def __init__(self, balance):

        # instance variable create karna
        self.balance = balance

    # balance check method create karna
    def check(self):

        # balance print karna
        print("Balance :", self.balance)

# object create karna
a = Atm(50000)

# method call karna
a.check()
📊 Output:

Balance : 50000
