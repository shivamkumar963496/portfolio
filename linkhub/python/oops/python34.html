# Ek Python program likho jo static method ka use karke
# message print kare.

# class create karna
class Demo:

    # static method create karna
    @staticmethod
    def show():

        # message print karna
        print("Static Method")

# static method call karna
Demo.show()
📊 Output:

Static Method
