# Question 22:
# Ek Python program likho jo encapsulation ka use karke
# private variable ko getter method se access kare.

# class create karna
class Student:

    # constructor create karna
    def __init__(self):

        # private variable create karna
        self.__marks = 90

    # getter method create karna
    def get_marks(self):

        # private variable return karna
        return self.__marks

# object create karna
s = Student()

# getter method call karna
print(s.get_marks())
📊 Output:

90
