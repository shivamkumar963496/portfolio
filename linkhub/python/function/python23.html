# Question 23:
# Ek Python program likho jo function aur return keyword ka use karke
# do numbers me maximum number find kare.

# function create karna
def maximum(a, b):

    # maximum number check karna
    if a > b:

        return a

    else:

        return b

# user se numbers input lena
a = int(input("Enter 1st Number : "))
b = int(input("Enter 2nd Number : "))

# function call karke result store karna
number = maximum(a, b)

# final output print karna
print("Maximum Number Is :", number)
📊 Output:

Enter 1st Number : 50
Enter 2nd Number : 80
Maximum Number Is : 80
