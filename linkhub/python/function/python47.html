# Ek Python program likho jo function ka use karke
# list ko ascending order me sort kare.

# function create karna
def sortlst(lst):

    # list ko sort karna
    lst.sort()

    # sorted list print karna
    print("After Sort :", lst)

# list create karna
lst = [1, 5, 7, 9, 6, 5, 4, 7, 8, 3, 4, 5]

# function call karna
sortlst(lst)
📊 Output:

After Sort : [1, 3, 4, 4, 5, 5, 5, 6, 7, 7, 8, 9]
