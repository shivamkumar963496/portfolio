# Ek Python program likho jo class method ka use karke
# rectangle ka area calculate kare.

# class create karna
class Rectangle:

    # method create karna
    def area(self, l, b):

        # rectangle ka area print karna
        print("Area of the Rectangle :", l * b)

# user se length input lena
l = int(input("Enter Length of the Rectangle : "))

# user se breadth input lena
b = int(input("Enter Breadth of the Rectangle : "))

# object create karna
r = Rectangle()

# method call karna
r.area(l, b)
📊 Output:

Enter Length of the Rectangle : 10
Enter Breadth of the Rectangle : 5
Area of the Rectangle : 50
