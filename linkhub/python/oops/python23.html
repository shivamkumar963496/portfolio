# Ek Python program likho jo encapsulation ka use karke
# setter aur getter method se marks update aur access kare.

# class create karna
class Student:

    # constructor create karna
    def __init__(self):

        # private variable create karna
        self.__marks = 0

    # setter method create karna
    def set_marks(self, m):

        # private variable update karna
        self.__marks = m

    # getter method create karna
    def get_marks(self):

        # private variable return karna
        return self.__marks

# object create karna
s = Student()

# setter method call karke marks set karna
s.set_marks(95)

# getter method call karke marks print karna
print(s.get_marks())
📊 Output:

95
