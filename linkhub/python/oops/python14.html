# Question 14:
# Ek Python program likho jo inheritance ka use karke
# parent aur child class methods ko call kare.

# parent class create karna
class Animal:

    # parent class method create karna
    def sound(self):

        # animal sound print karna
        print("Animal Sound")

# child class create karna
class Dog(Animal):

    # child class method create karna
    def bark(self):

        # dog bark sound print karna
        print("Dog Bark")

# object create karna
d = Dog()

# parent class method call karna
d.sound()

# child class method call karna
d.bark()
📊 Output:

Animal Sound
Dog Bark
