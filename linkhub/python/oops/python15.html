# Question 15:
# Ek Python program likho jo multilevel inheritance ka use karke
# teen classes ke methods ko access kare.

# first parent class create karna
class A:

    # method create karna
    def showA(self):

        # message print karna
        print("A")

# second class create karna
class B(A):

    # method create karna
    def showB(self):

        # message print karna
        print("B")

# third class create karna
class C(B):

    # method create karna
    def showC(self):

        # message print karna
        print("C")

# object create karna
obj = C()

# methods call karna
obj.showA()
obj.showB()
obj.showC()
📊 Output:

A
B
C
