# Question 17:
# Ek Python program likho jo hierarchical inheritance ka use karke
# ek parent class ke method ko multiple child classes se access kare.

# parent class create karna
class Father:

    # parent class method create karna
    def show(self):

        # message print karna
        print("Father")

# first child class create karna
class Child1(Father):

    pass

# second child class create karna
class Child2(Father):

    pass

# first object create karna
c1 = Child1()

# second object create karna
c2 = Child2()

# methods call karna
c1.show()
c2.show()
📊 Output:

Father
Father
