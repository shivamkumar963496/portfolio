# Question 14:
# Ek Python program likho jo function ka use karke
# kisi number ka table print kare.

# function create karna
def table(n):

    # loop se table print karna
    for i in range(1, 11):

        print(n, "X", i, "=", n * i)

# heading print karna
print("===== PRINT THE TABLE OF ANY NUMBER =====")

# user se number input lena
n = int(input("Enter a Number : "))

# function call karna
table(n)
