# Question 24:
# Ek Python program likho jo polymorphism ka use karke
# alag-alag classes ke same method ko call kare.

# first class create karna
class Bird:

    # method create karna
    def sound(self):

        # bird sound print karna
        print("Bird Fly")

# second class create karna
class Dog:

    # method create karna
    def sound(self):

        # dog sound print karna
        print("Dog Bark")

# loop ke through methods call karna
for obj in [Bird(), Dog()]:

    obj.sound()
📊 Output:

Bird Fly
Dog Bark
