# Ek Python program likho jo constructor aur class ka use karke
# check kare ki candidate voting ke liye eligible hai ya nahi.

# class create karna
class Vote:

    # constructor create karna
    def __init__(self, age):

        # instance variable create karna
        self.age = age

    # method create karna
    def check(self):

        # voting eligibility check karna
        if self.age >= 18:

            print("Eligible For Voting")

        else:

            print("Not Eligible For Voting")

# user se age input lena
age = int(input("Enter Candidate Age : "))

# object create karna
v = Vote(age)

# method call karna
v.check()
📊 Output:

Enter Candidate Age : 20
Eligible For Voting
