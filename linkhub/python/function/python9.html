# Ek Python program likho jo function ka use karke
# kisi number ko check kare ki vo Even hai ya Odd.

# function create karna
def check():

    # user se number input lena
    n = int(input("Enter Number : "))

    # even ya odd check karna
    if n % 2 == 0:
        print("EVEN")

    else:
        print("ODD")

# heading print karna
print("====== CHECK ODD OR EVEN ======")

# function call karna
check()
