# Ek Python program likho jo class method ka use karke
# circle ka area calculate kare.

# class create karna
class Area:

    # circle area method create karna
    def circle(self, r):

        # circle ka area print karna
        print("Area of the Circle :", 3.1415 * r * r)

# object create karna
a = Area()

# method call karna
a.circle(5)
📊 Output:

Area of the Circle : 78.5375
