# Question 32:
# Ek Python program likho jo constructor ka use karke
# mobile ki brand aur price print kare.

# class create karna
class Mobile:

    # constructor create karna
    def __init__(self, brand, price):

        # instance variables create karna
        self.brand = brand
        self.price = price

# object create karna
m = Mobile("Samsung", 20000)

# mobile brand print karna
print(m.brand)

# mobile price print karna
print(m.price)
📊 Output:

Samsung
20000
