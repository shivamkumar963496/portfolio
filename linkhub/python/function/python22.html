# Ek Python program likho jo function aur return keyword ka use karke
# kisi number ka square calculate kare.

# function create karna
def square(n):

    # square return karna
    return n * n

# user se number input lena
n = int(input("Enter a Number : "))

# function call karke result print karna
print(square(n))
