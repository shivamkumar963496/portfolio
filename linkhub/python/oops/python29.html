# ==========================================
# Bank Deposit Management System
# ==========================================


# class create karna
class Bank:

    # ======================================
    # Constructor
    # ======================================

    def __init__(self, balance):

        # instance variable create karna
        self.balance = balance

        # account creation message print karna
        print("\n========== BANK ACCOUNT CREATED ==========\n")

        print("Current Balance :", self.balance)


    # ======================================
    # Deposit Method
    # ======================================

    def deposit(self, amount):

        # deposit amount check karna
        if amount > 0:

            # balance update karna
            self.balance += amount

            # success message print karna
            print("\n========== DEPOSIT SUCCESSFUL ==========\n")

            print("Deposit Amount :", amount)

            print("Updated Balance :", self.balance)

        else:

            # invalid amount message print karna
            print("\n========== INVALID DEPOSIT AMOUNT ==========\n")


# ==========================================
# Create Bank Account
# ==========================================

# object create karna
b = Bank(5000)


# ==========================================
# User Input
# ==========================================

# user se deposit amount input lena
amount = int(input("\nEnter Deposit Amount : "))


# ==========================================
# Deposit Money
# ==========================================

# deposit method call karna
b.deposit(amount)
