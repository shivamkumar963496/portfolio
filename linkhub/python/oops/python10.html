# Question 10:
# Ek Python program likho jo constructor ka use karke
# employee ki details print kare.

# class create karna
class Employee:

    # constructor create karna
    def __init__(self, name, salary, company):

        # instance variables create karna
        self.name = name
        self.salary = salary
        self.company = company

# object create karna
e = Employee("Shivam", 50000, "Google")

# employee details print karna
print("Name    :", e.name)
print("Salary  :", e.salary)
print("Company :", e.company)
📊 Output:

Name : Shivam
Salary : 50000
Company : Google
