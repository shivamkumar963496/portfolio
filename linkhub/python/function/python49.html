# Ek Python program likho jo function ka use karke
# username aur password check kare.

# function create karna
def login(user, password):

    # username aur password verify karna
    if user == "admin" and password == "1234":

        print("Login Success")

    else:

        print("Login Failed")

# user se username input lena
user = input("Enter User Name : ")

# user se password input lena
password = input("Enter Password : ")

# function call karna
login(user, password)
📊 Output:

Enter User Name : admin
Enter Password : 1234
Login Success
