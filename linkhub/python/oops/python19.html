# Question 19:
# Ek Python program likho jo inheritance aur super() method ka use karke
# parent aur child constructor ko call kare.

# parent class create karna
class Father:

    # parent constructor create karna
    def __init__(self):

        # parent constructor message print karna
        print("Parent Constructor")

# child class create karna
class Child(Father):

    # child constructor create karna
    def __init__(self):

        # parent constructor call karna
        super().__init__()

        # child constructor message print karna
        print("Child Constructor")

# object create karna
c = Child()
📊 Output:

Parent Constructor
Child Constructor
