# Ek Python program likho jo function ka use karke
# check kare ki number Prime hai ya nahi.

# function create karna
def prime(n):

    # counter variable initialize karna
    c = 0

    # factors count karna
    for i in range(1, n + 1):

        if n % i == 0:

            c += 1

    # prime condition check karna
    if c == 2:

        print("PRIME NUMBER")

    else:

        print("NOT PRIME NUMBER")

# heading print karna
print("===== CHECK NUMBER PRIME OR NOT =====")

# user se number input lena
n = int(input("Enter a Number : "))

# function call karna
prime(n)
