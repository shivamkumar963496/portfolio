# Ek Python program likho jo function ka use karke
# 3 numbers ka average calculate kare.

# function create karna
def avg(a, b, c):

    # average calculate karna
    print((a + b + c) / 3)

# heading print karna
print("\n===== CALCULATE THE AVERAGE OF 3 GIVEN NUMBERS =====\n")

# user se input lena
a = int(input("Enter the Value of A : "))
b = int(input("Enter the Value of B : "))
c = int(input("Enter the Value of C : "))

# function call karna
avg(a, b, c)
