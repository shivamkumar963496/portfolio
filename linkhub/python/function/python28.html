# Ek Python program likho jo recursion function ka use karke
# Fibonacci series ka nth term print kare.

# recursive function create karna
def fib(n):

    # base condition check karna
    if n <= 1:

        return n

    else:

        # fibonacci calculate karna
        return fib(n - 1) + fib(n - 2)

# user se number input lena
n = int(input("Enter a Number : "))

# fibonacci term print karna
print(fib(n))
