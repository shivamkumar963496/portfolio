# Question 26:
# Ek Python program likho jo abstract class aur abstract method ka use karke
# vehicle start method implement kare.

# abstract class import karna
from abc import ABC, abstractmethod

# abstract class create karna
class Vehicle(ABC):

    # abstract method create karna
    @abstractmethod
    def start(self):

        pass

# child class create karna
class Car(Vehicle):

    # abstract method implement karna
    def start(self):

        # message print karna
        print("Car Start")

# object create karna
c = Car()

# method call karna
c.start()
📊 Output:

Car Start
