# check kare ki diya gaya text Palindrome hai ya nahi.

# function create karna
def palindrome(s):

    # palindrome check karna
    if s == s[::-1]:

        # reversed text print karna
        print("After Reverse :", s[::-1])

        # palindrome message print karna
        print("Palindrome")

    else:

        # reversed text print karna
        print("After Reverse :", s[::-1])

        # not palindrome message print karna
        print("Not Palindrome")

# user se text input lena
text = input("Enter Any Text : ")

# function call karna
palindrome(text)
📊 Output:

Enter Any Text : madam
After Reverse : madam
Palindrome
