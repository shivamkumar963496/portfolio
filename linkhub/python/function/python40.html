# Question 40:
# Ek Python program likho jo nested function ka use karke
# outer aur inner function ko call kare.

# outer function create karna
def outer():

    # outer function message print karna
    print("Outer Function")

    # inner function create karna
    def inner():

        # inner function message print karna
        print("Inner Function")

    # inner function call karna
    inner()

# outer function call karna
outer()
📊 Output:

Outer Function
Inner Function
