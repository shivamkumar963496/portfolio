# Ek Python program likho jo **kwargs ka use karke
# multiple data ko dictionary ke form me print kare.

# function create karna
def data(**x):

    # dictionary data print karna
    print("Data Is :", x)

# function call karna
data(name = "Shivam", age = 20, college = "NIET")
📊 Output:

Data Is : {'name': 'Shivam', 'age': 20, 'college': 'NIET'}
