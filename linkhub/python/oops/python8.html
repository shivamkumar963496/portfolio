# Ek Python program likho jo class method ka use karke
# do numbers ka addition kare.

# class create karna
class Add:

    # method create karna
    def sum(self, a, b):

        # addition print karna
        print(a + b)

# user se input lena
a = int(input("Enter 1st Number : "))
b = int(input("Enter 2nd Number : "))

# object create karna
obj = Add()

# method call karna
obj.sum(a, b)
