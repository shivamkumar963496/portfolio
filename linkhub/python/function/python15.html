# Ek Python program likho jo function ka use karke
# kisi number ka factorial calculate kare.

# function create karna
def fact(n):

    # factorial variable initialize karna
    f = 1

    # loop se factorial calculate karna
    for i in range(1, n + 1):

        f = f * i

    # factorial print karna
    print(f)

# heading print karna
print("\n===== CALCULATE THE FACTORIAL OF NUMBER =====\n")

# user se number input lena
n = int(input("Enter a Number : "))

# function call karna
fact(n)
