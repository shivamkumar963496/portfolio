# ==========================================
# College Students Record Management System
# ==========================================


# ==========================================
# Student Class
# ==========================================

# class create karna
class Student:

    # constructor create karna
    def __init__(self, name, age, course, college):

        # instance variables create karna
        self.name = name
        self.age = age
        self.course = course
        self.college = college


# ==========================================
# Display Heading
# ==========================================

print("\n========== COLLEGE STUDENTS RECORD ==========\n")


# ==========================================
# Student 1 Record
# ==========================================

# first object create karna
student1 = Student("Shivam", 21, "Polytechnic", "NIET")

print("========== STUDENT - 1 RECORD ==========")

print("Name    :", student1.name)
print("Age     :", student1.age)
print("Course  :", student1.course)
print("College :", student1.college)


# ==========================================
# Student 2 Record
# ==========================================

# second object create karna
student2 = Student("Yograj", 22, "BTech", "NIET")

print("\n========== STUDENT - 2 RECORD ==========")

print("Name    :", student2.name)
print("Age     :", student2.age)
print("Course  :", student2.course)
print("College :", student2.college)


# ==========================================
# Student 3 Record
# ==========================================

# third object create karna
student3 = Student("Rohit", 19, "DPharma", "NCP")

print("\n========== STUDENT - 3 RECORD ==========")

print("Name    :", student3.name)
print("Age     :", student3.age)
print("Course  :", student3.course)
print("College :", student3.college)
