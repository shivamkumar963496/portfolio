# Ek Python program likho jo recursion function ka use karke
# kisi number ka power calculate kare.

# recursive function create karna
def power(a, b):

    # base condition check karna
    if b == 0:

        return 1

    else:

        # recursive power calculate karna
        return a * power(a, b - 1)

# user se base input lena
m = int(input("Enter the Base : "))

# user se power input lena
n = int(input("Enter the Power : "))

# power calculate karna
result = power(m, n)

# final output print karna
print("Result :", result)
📊 Output:

Enter the Base : 2
Enter the Power : 5
Result : 32
