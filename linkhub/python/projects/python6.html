🧮 Python Mini Project 6 - Advanced Calculator

# -----------------------------------------------------------
# Question:
# Python me ek Simple Calculator program likhiye jo user se
# do numbers le aur Addition, Subtraction, Multiplication
# aur Division perform kare.
# Program menu driven ho aur loop ka use kare.
# -----------------------------------------------------------

# calculator heading display karna
print("\n===== PYTHON CALCULATOR =====")

# loop tab tak chalega jab tak user exit na kare
while True:

    # menu display karna
    print("\n1. Addition")
    print("2. Subtraction")
    print("3. Multiplication")
    print("4. Division")
    print("5. Exit")

    # user se choice lena
    choice = int(input("Enter Your Choice (1-5) : "))

    # Exit Program
    if choice == 5:

        print("========= Calculator Closed =========")
        break

    # Valid choice check karna
    if choice == 1 or choice == 2 or choice == 3 or choice == 4:

        num1 = int(input("Enter 1st Number : "))
        num2 = int(input("Enter 2nd Number : "))

        # Addition
        if choice == 1:

            result = num1 + num2
            print("Addition :", result)

        # Subtraction
        elif choice == 2:

            result = num1 - num2
            print("Subtraction :", result)

        # Multiplication
        elif choice == 3:

            result = num1 * num2
            print("Multiplication :", result)

        # Division
        elif choice == 4:

            if num2 == 0:

                print("Division Not Allowed By 0")

            else:

                result = num1 / num2
                print("Division :", result)

    else:

        print("Invalid Input")
