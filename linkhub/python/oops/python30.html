# ==========================================
# Bank Withdrawal Management System
# ==========================================


# class create karna
class Bank:

    # ======================================
    # Constructor
    # ======================================

    def __init__(self, balance):

        # instance variable create karna
        self.balance = balance

        # account creation message print karna
        print("\n========== BANK ACCOUNT CREATED ==========\n")

        print("Current Balance :", self.balance)


    # ======================================
    # Withdrawal Method
    # ======================================

    def withdraw(self, amount):

        # invalid amount check karna
        if amount <= 0:

            print("\nInvalid Withdrawal Amount")


        # sufficient balance check karna
        elif amount <= self.balance:

            # balance update karna
            self.balance -= amount

            # success message print karna
            print("\n========== WITHDRAWAL SUCCESSFUL ==========\n")

            print("Withdrawal Amount :", amount)

            print("Remaining Balance :", self.balance)


        else:

            # insufficient balance message print karna
            print("\n========== TRANSACTION FAILED ==========\n")

            print("Insufficient Balance")


# ==========================================
# Create Bank Account
# ==========================================

# object create karna
bank_account = Bank(5000)


# ==========================================
# User Input
# ==========================================

# user se withdrawal amount input lena
amount = int(input("\nEnter Withdrawal Amount : "))


# ==========================================
# Withdraw Money
# ==========================================

# withdraw method call karna
bank_account.withdraw(amount)
