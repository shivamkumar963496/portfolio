# Ek Python program likho jo function aur global keyword ka use karke
# bank account se withdrawal kare.

# global balance variable create karna
balance = 10000

# function create karna
def withdrawal(amount):

    # global variable access karna
    global balance

    # balance update karna
    balance = balance - amount

    # remaining balance print karna
    print("Remaining Balance After Withdrawal :", balance)

# current balance print karna
print("Total Balance in Your Account :", balance)

# user se withdrawal amount input lena
amount = int(input("Enter Withdrawal Amount : "))

# function call karna
withdrawal(amount)
📊 Output:

Total Balance in Your Account : 10000
Enter Withdrawal Amount : 2500
Remaining Balance After Withdrawal : 7500
