# Question 4:
# Ek Python program likho jo constructor aur object ka use karke
# alag-alag students ke names print kare.

# class create karna
class Student:

    # constructor create karna
    def __init__(self, name):

        # instance variable create karna
        self.name = name

# first object create karna
obj1 = Student("Shivam")

# second object create karna
obj2 = Student("Yograj")

# first student ka name print karna
print(obj1.name)

# second student ka name print karna
print(obj2.name)
📊 Output:

Shivam
Yograj
