# Question 17:
# Ek Python program likho jo custom exception create kare.

# custom exception class create karna
class AgeError(Exception):
    pass

# try block create karna
try:

    # user se age input lena
    age = int(input("Enter Your Age : "))

    # condition check karna
    if age < 18:

        # custom exception raise karna
        raise AgeError("Age Must Be 18+")

# custom exception handle karna
except AgeError as e:

    # error message print karna
    print(e)

Enter Your Age : 15
