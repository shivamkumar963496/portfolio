# Question 18:
# Ek Python program likho jo method overriding ka use karke
# child class method ko parent class method par override kare.

# parent class create karna
class Father:

    # parent class method create karna
    def show(self):

        # parent method message print karna
        print("Parent Method")

# child class create karna
class Child(Father):

    # same method override karna
    def show(self):

        # child method message print karna
        print("Child Method")

# object create karna
c = Child()

# overridden method call karna
c.show()
📊 Output:

Child Method
