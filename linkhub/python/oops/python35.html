# Question 35:
# Ek Python program likho jo class method ka use karke
# class variable ki value print kare.

# class create karna
class Demo:

    # class variable create karna
    x = 10

    # class method create karna
    @classmethod
    def show(cls):

        # class variable print karna
        print(cls.x)

# class method call karna
Demo.show()
📊 Output:

10
