# Question 20:
# Ek Python program likho jo encapsulation ka use karke
# private variable ko access kare.

# class create karna
class Bank:

    # constructor create karna
    def __init__(self):

        # private variable create karna
        self.__balance = 5000

    # method create karna
    def show(self):

        # private variable print karna
        print(self.__balance)

# object create karna
o = Bank()

# method call karna
o.show()
📊 Output:

5000
