# Question 16:
# Ek Python program likho jo multiple inheritance ka use karke
# father aur mother class ke methods ko access kare.

# father class create karna
class Father:

    # father class method create karna
    def home(self):

        # message print karna
        print("Home")

# mother class create karna
class Mother:

    # mother class method create karna
    def car(self):

        # message print karna
        print("Car")

# child class create karna
class Child(Father, Mother):

    pass

# object create karna
c = Child()

# father class method call karna
c.home()

# mother class method call karna
c.car()
📊 Output:

Home
Car
